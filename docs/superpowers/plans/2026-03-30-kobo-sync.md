# Kobo Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Emulate the Kobo Store API so Kobo e-readers sync with Verso automatically — library discovery, book download (with KEPUB conversion), bidirectional reading progress, and transparent proxy to the real Kobo store.

**Architecture:** Fastify routes at `/kobo/:token/*` implementing Kobo's proprietary sync protocol. Token-based auth (URL contains per-user token). Snapshot-based library diffing for incremental sync. Calibre for EPUB→KEPUB conversion. Transparent proxy to `storeapi.kobo.com` for unhandled requests. tRPC router for settings UI.

**Tech Stack:** Fastify, Drizzle ORM, tRPC, Calibre CLI (`ebook-convert`), Node.js `fetch` (undici), React

```typescript
import type { FastifyRequest, FastifyReply } from "fastify";
import { eq } from "drizzle-orm";
import { koboSettings, users } from "@verso/shared";
import type { AppDatabase } from "../db/client.js";

declare module "fastify" {
  interface FastifyRequest {
    koboUser?: { id: string; email: string; token: string; convertToKepub: boolean };
  }
}

export function createKoboAuthHook(db: AppDatabase) {
  return async (req: FastifyRequest<{ Params: { token: string } }>, reply: FastifyReply) => {
    const { token } = req.params;

    if (!token) {
      return reply.code(401).send({ message: "Missing token" });
    }

    const settings = await db
      .select({
        userId: koboSettings.userId,
        token: koboSettings.token,
        convertToKepub: koboSettings.convertToKepub,
        enabled: koboSettings.enabled,
      })
      .from(koboSettings)
      .where(eq(koboSettings.token, token))
      .get();

    if (!settings || !settings.enabled) {
      return reply.code(401).send({ message: "Invalid or disabled sync token" });
    }

    const user = await db
      .select({ id: users.id, email: users.email })
      .from(users)
      .where(eq(users.id, settings.userId))
      .get();

    if (!user) {
      return reply.code(401).send({ message: "User not found" });
    }

    req.koboUser = {
      id: user.id,
      email: user.email,
      token: settings.token,
      convertToKepub: settings.convertToKepub ?? true,
    };
  };
}
```

```typescript
const KOBO_STORE_API = "https://storeapi.kobo.com";

export async function proxyToKobo(
  method: string,
  path: string,
  headers: Record<string, string | undefined>,
  body?: unknown,
): Promise<{ status: number; headers: Record<string, string>; body: unknown }> {
  const url = `${KOBO_STORE_API}${path}`;

  const forwardHeaders: Record<string, string> = {};
  // Forward relevant headers, strip auth/host
  for (const [key, value] of Object.entries(headers)) {
    const lower = key.toLowerCase();
    if (lower === "host" || lower === "authorization" || lower === "content-length") continue;
    if (value) forwardHeaders[key] = value;
  }

  try {
    const response = await fetch(url, {
      method,
      headers: forwardHeaders,
      body: body && method !== "GET" && method !== "HEAD" ? JSON.stringify(body) : undefined,
      signal: AbortSignal.timeout(15_000),
    });

    const responseHeaders: Record<string, string> = {};
    response.headers.forEach((value, key) => {
      responseHeaders[key] = value;
    });

    let responseBody: unknown;
    const contentType = response.headers.get("content-type") || "";
    if (contentType.includes("json")) {
      responseBody = await response.json();
    } else {
      responseBody = await response.text();
    }

    return { status: response.status, headers: responseHeaders, body: responseBody };
  } catch (err: any) {
    if (err.name === "TimeoutError") {
      return { status: 504, headers: {}, body: { message: "Upstream timeout" } };
    }
    return { status: 502, headers: {}, body: { message: "Proxy error" } };
  }
}
```

```typescript
import { execFile as execFileCb } from "node:child_process";
import { promisify } from "node:util";
import { existsSync } from "node:fs";
import { stat } from "node:fs/promises";
import type { StorageService } from "./storage.js";
import type { Config } from "../config.js";

const execFile = promisify(execFileCb);

function calibrePath(config: Config): string {
  return config.CALIBRE_PATH ? `${config.CALIBRE_PATH}/ebook-convert` : "ebook-convert";
}

/**
 * Convert EPUB to KEPUB, caching the result.
 * Returns the full filesystem path to the KEPUB file.
 * Falls back to the original EPUB path if conversion fails.
 */
export async function getKepubPath(
  bookId: string,
  epubRelativePath: string,
  bookUpdatedAt: string,
  storage: StorageService,
  config: Config,
): Promise<{ path: string; isKepub: boolean }> {
  const kepubRelativePath = epubRelativePath.replace(/\.epub$/i, ".kepub.epub");
  const kepubFullPath = storage.fullPath(kepubRelativePath);
  const epubFullPath = storage.fullPath(epubRelativePath);

  // Check if cached KEPUB exists and is newer than the book's updatedAt
  if (existsSync(kepubFullPath)) {
    try {
      const kepubStat = await stat(kepubFullPath);
      const kepubModified = kepubStat.mtime.toISOString();
      if (kepubModified >= bookUpdatedAt) {
        return { path: kepubFullPath, isKepub: true };
      }
    } catch {
      // Fall through to reconvert
    }
  }

  // Convert
  try {
    await execFile(
      calibrePath(config),
      [epubFullPath, kepubFullPath],
      { timeout: 120_000 },
    );
    return { path: kepubFullPath, isKepub: true };
  } catch (err) {
    console.warn(`KEPUB conversion failed for book ${bookId}, serving original EPUB:`, err);
    return { path: epubFullPath, isKepub: false };
  }
}
```

```typescript
import { eq, and, notInArray, inArray } from "drizzle-orm";
import { books, koboSyncSnapshots, readingProgress } from "@verso/shared";
import type { AppDatabase } from "../db/client.js";

type BookRow = typeof books.$inferSelect;

export interface Entitlement {
  NewEntitlement?: object;
  ChangedEntitlement?: object;
  ChangedReadingState?: object;
}

function buildBookMetadata(book: BookRow, token: string, entitlementId: string) {
  return {
    Categories: [],
    ContributorRoles: [{ Name: book.author }],
    CoverImageId: book.id,
    Description: book.description || "",
    DownloadUrls: [
      {
        Format: "EPUB",
        Size: book.fileSize,
        Url: `/kobo/${token}/v1/books/${book.id}/download`,
        Platform: "Generic",
      },
    ],
    EntitlementId: entitlementId,
    Language: book.language || "en",
    PublicationDate: book.year ? `${book.year}-01-01T00:00:00Z` : "2000-01-01T00:00:00Z",
    Publisher: { Name: book.publisher || "" },
    RevisionId: entitlementId,
    Series: book.series
      ? { Name: book.series, Number: book.seriesIndex ?? 0, NumberFloat: book.seriesIndex ?? 0 }
      : undefined,
    Title: book.title,
    WorkId: entitlementId,
  };
}

function buildBookEntitlement(book: BookRow, entitlementId: string) {
  return {
    Accessibility: "Full",
    ActivePeriod: { From: "2000-01-01T00:00:00Z" },
    Created: book.createdAt,
    CrossRevisionId: entitlementId,
    Id: entitlementId,
    IsHiddenFromArchive: false,
    IsLocked: false,
    IsRemoved: false,
    LastModified: book.updatedAt,
    OriginCategory: "Imported",
    RevisionId: entitlementId,
    Status: "Active",
  };
}

function buildReadingState(percentage: number | null, lastReadAt: string | null) {
  const pct = percentage ?? 0;
  return {
    LastModified: lastReadAt || new Date().toISOString(),
    CurrentBookmark: {
      Location: { Value: String(pct) },
      ProgressPercent: pct,
    },
    StatusInfo: {
      LastModified: lastReadAt || new Date().toISOString(),
      Status: pct >= 98 ? "Finished" : pct > 0 ? "Reading" : "ReadyToRead",
    },
  };
}

export async function computeSyncEntitlements(
  db: AppDatabase,
  userId: string,
  token: string,
): Promise<{ entitlements: Entitlement[]; hasMore: boolean }> {
  const BATCH_SIZE = 100;

  // Get all books
  const allBooks = await db.select().from(books).all();

  // Get existing snapshots for this user
  const snapshots = await db
    .select()
    .from(koboSyncSnapshots)
    .where(eq(koboSyncSnapshots.userId, userId))
    .all();

  const snapshotMap = new Map(snapshots.map((s) => [s.bookId, s]));
  const bookMap = new Map(allBooks.map((b) => [b.id, b]));

  // Get reading progress for this user
  const progressRows = await db
    .select()
    .from(readingProgress)
    .where(eq(readingProgress.userId, userId))
    .all();
  const progressMap = new Map(progressRows.map((p) => [p.bookId, p]));

  const entitlements: Entitlement[] = [];

  // New books (in library but no snapshot)
  for (const book of allBooks) {
    if (snapshotMap.has(book.id)) continue;

    const entitlementId = crypto.randomUUID();
    const progress = progressMap.get(book.id);

    entitlements.push({
      NewEntitlement: {
        BookEntitlement: buildBookEntitlement(book, entitlementId),
        BookMetadata: buildBookMetadata(book, token, entitlementId),
        ReadingState: buildReadingState(progress?.percentage ?? null, progress?.lastReadAt ?? null),
      },
    });

    // Create snapshot
    await db.insert(koboSyncSnapshots).values({
      userId,
      bookId: book.id,
      entitlementId,
      lastModified: book.updatedAt,
    });

    if (entitlements.length >= BATCH_SIZE) break;
  }

  // Changed books (snapshot exists but book updated)
  if (entitlements.length < BATCH_SIZE) {
    for (const snapshot of snapshots) {
      const book = bookMap.get(snapshot.bookId);
      if (!book) continue; // Deleted — handle below
      if (book.updatedAt === snapshot.lastModified) continue;

      const progress = progressMap.get(book.id);

      entitlements.push({
        ChangedEntitlement: {
          BookEntitlement: buildBookEntitlement(book, snapshot.entitlementId),
          BookMetadata: buildBookMetadata(book, token, snapshot.entitlementId),
          ReadingState: buildReadingState(progress?.percentage ?? null, progress?.lastReadAt ?? null),
        },
      });

      // Update snapshot
      await db
        .update(koboSyncSnapshots)
        .set({ lastModified: book.updatedAt })
        .where(eq(koboSyncSnapshots.id, snapshot.id));

      if (entitlements.length >= BATCH_SIZE) break;
    }
  }

  // Deleted books (snapshot exists but book gone)
  if (entitlements.length < BATCH_SIZE) {
    for (const snapshot of snapshots) {
      if (bookMap.has(snapshot.bookId)) continue;

      entitlements.push({
        ChangedReadingState: {
          EntitlementId: snapshot.entitlementId,
          StatusInfo: { Status: "RevisionChanged" },
        },
      });

      // Remove snapshot
      await db
        .delete(koboSyncSnapshots)
        .where(eq(koboSyncSnapshots.id, snapshot.id));

      if (entitlements.length >= BATCH_SIZE) break;
    }
  }

  // Check if there's more to sync
  const remainingNewBooks = allBooks.filter((b) => !snapshotMap.has(b.id)).length;
  const remainingChanged = snapshots.filter((s) => {
    const book = bookMap.get(s.bookId);
    return book && book.updatedAt !== s.lastModified;
  }).length;
  const remainingDeleted = snapshots.filter((s) => !bookMap.has(s.bookId)).length;
  const totalRemaining = remainingNewBooks + remainingChanged + remainingDeleted;
  const hasMore = totalRemaining > entitlements.length;

  return { entitlements, hasMore };
}
```

```typescript
import type { FastifyInstance } from "fastify";
import { eq, and } from "drizzle-orm";
import { createReadStream } from "node:fs";
import { createKoboAuthHook } from "../middleware/kobo-auth.js";
import { proxyToKobo } from "../services/kobo-proxy.js";
import { computeSyncEntitlements } from "../services/kobo-sync.js";
import { getKepubPath } from "../services/kobo-convert.js";
import { books, readingProgress, devices, readingSessions } from "@verso/shared";
import type { AppDatabase } from "../db/client.js";
import type { StorageService } from "../services/storage.js";
import type { Config } from "../config.js";

export function registerKoboRoutes(
  app: FastifyInstance,
  db: AppDatabase,
  storage: StorageService,
  config: Config,
) {
  const authHook = createKoboAuthHook(db);

  // POST /kobo/:token/v1/auth/device — synthetic auth
  app.post<{ Params: { token: string } }>(
    "/kobo/:token/v1/auth/device",
    { preHandler: authHook },
    async (req, reply) => {
      return reply.send({
        AccessToken: crypto.randomUUID(),
        RefreshToken: crypto.randomUUID(),
        TokenType: "Bearer",
        TrackingId: crypto.randomUUID(),
        UserKey: req.koboUser!.id,
      });
    },
  );

  // GET /kobo/:token/v1/initialization — proxy + rewrite URLs
  app.get<{ Params: { token: string } }>(
    "/kobo/:token/v1/initialization",
    { preHandler: authHook },
    async (req, reply) => {
      const token = req.koboUser!.token;
      const result = await proxyToKobo("GET", "/v1/initialization", req.headers as Record<string, string>);

      if (result.status !== 200 || typeof result.body !== "object" || !result.body) {
        return reply.code(result.status).send(result.body);
      }

      const body = result.body as any;

      // Determine our base URL from the request
      const proto = req.headers["x-forwarded-proto"] || req.protocol;
      const host = req.headers["x-forwarded-host"] || req.hostname;
      const baseUrl = `${proto}://${host}`;

      // Rewrite resource URLs
      if (body.Resources) {
        for (const resource of Object.values(body.Resources) as any[]) {
          if (resource?.href?.includes("storeapi.kobo.com")) {
            resource.href = resource.href.replace(
              /https?:\/\/storeapi\.kobo\.com/,
              `${baseUrl}/kobo/${token}`,
            );
          }
        }
        // Override specific resources
        if (body.Resources.image_host) {
          body.Resources.image_host = baseUrl;
        }
        if (body.Resources.image_url_template) {
          body.Resources.image_url_template = `${baseUrl}/kobo/${token}/v1/books/{ImageId}/metadata/image/{Width}/{Height}/{Quality}/{IsGreyscale}`;
        }
        if (body.Resources.image_url_quality_template) {
          body.Resources.image_url_quality_template = `${baseUrl}/kobo/${token}/v1/books/{ImageId}/metadata/image/{Width}/{Height}/{Quality}/{IsGreyscale}`;
        }
      }

      return reply.send(body);
    },
  );

  // GET /kobo/:token/v1/library/sync — main sync endpoint
  app.get<{ Params: { token: string } }>(
    "/kobo/:token/v1/library/sync",
    { preHandler: authHook },
    async (req, reply) => {
      const userId = req.koboUser!.id;
      const token = req.koboUser!.token;

      const { entitlements, hasMore } = await computeSyncEntitlements(db, userId, token);

      if (hasMore) {
        reply.header("X-Kobo-Sync", "continue");
      }

      return reply.send(entitlements);
    },
  );

  // GET /kobo/:token/v1/books/:bookId/download — serve book file
  app.get<{ Params: { token: string; bookId: string } }>(
    "/kobo/:token/v1/books/:bookId/download",
    { preHandler: authHook },
    async (req, reply) => {
      const { bookId } = req.params;

      const book = await db
        .select()
        .from(books)
        .where(eq(books.id, bookId))
        .get();

      if (!book) {
        return reply.code(404).send({ message: "Book not found" });
      }

      const wantKepub = req.koboUser!.convertToKepub && book.fileFormat === "epub";

      if (wantKepub) {
        const { path: filePath, isKepub } = await getKepubPath(
          book.id,
          book.filePath,
          book.updatedAt,
          storage,
          config,
        );

        const ext = isKepub ? "kepub.epub" : "epub";
        reply.header("Content-Type", "application/epub+zip");
        reply.header("Content-Disposition", `attachment; filename="${book.title}.${ext}"`);
        return reply.send(createReadStream(filePath));
      }

      // Serve original file
      const stream = storage.stream(book.filePath);
      reply.header("Content-Type", "application/octet-stream");
      reply.header("Content-Disposition", `attachment; filename="${book.title}.${book.fileFormat}"`);
      return reply.send(stream);
    },
  );

  // PUT /kobo/:token/v1/library/:bookId/state — reading progress sync
  app.put<{ Params: { token: string; bookId: string }; Body: any }>(
    "/kobo/:token/v1/library/:bookId/state",
    { preHandler: authHook },
    async (req, reply) => {
      const { bookId } = req.params;
      const userId = req.koboUser!.id;
      const body = req.body as any;
      const now = new Date().toISOString();

      const book = await db
        .select({ id: books.id })
        .from(books)
        .where(eq(books.id, bookId))
        .get();

      if (!book) {
        return reply.code(404).send({ message: "Book not found" });
      }

      // Extract reading state
      const readingState = body?.ReadingStates?.[0];
      const percentage = readingState?.CurrentBookmark?.ProgressPercent ?? 0;
      const status = readingState?.StatusInfo?.Status;

      // Upsert device — use a deterministic ID from headers or generate one
      const koboDeviceId = (req.headers["x-kobo-deviceid"] as string) || `kobo-${userId}`;
      const existingDevice = await db.select().from(devices).where(eq(devices.id, koboDeviceId)).get();
      if (existingDevice) {
        await db.update(devices).set({ lastSeen: now }).where(eq(devices.id, koboDeviceId));
      } else {
        await db.insert(devices).values({
          id: koboDeviceId,
          userId,
          model: "Kobo",
          lastSeen: now,
        });
      }

      // Upsert reading progress
      const existing = await db
        .select()
        .from(readingProgress)
        .where(and(eq(readingProgress.userId, userId), eq(readingProgress.bookId, bookId)))
        .get();

      const finishedAt = status === "Finished" || percentage >= 98 ? now : null;

      if (existing) {
        await db.update(readingProgress).set({
          percentage,
          lastReadAt: now,
          deviceId: koboDeviceId,
          finishedAt: existing.finishedAt ?? finishedAt,
        }).where(eq(readingProgress.id, existing.id));
      } else {
        await db.insert(readingProgress).values({
          userId,
          bookId,
          percentage,
          startedAt: now,
          lastReadAt: now,
          deviceId: koboDeviceId,
          finishedAt,
        });
      }

      // Create reading session
      await db.insert(readingSessions).values({
        userId,
        bookId,
        startedAt: now,
        endedAt: now,
        durationMinutes: 0,
        deviceId: koboDeviceId,
        source: "kobo",
        bookTitle: null,
      });

      return reply.send({});
    },
  );

  // GET /kobo/:token/v1/books/:bookId/metadata/image/* — serve covers
  app.get<{ Params: { token: string; bookId: string; "*": string } }>(
    "/kobo/:token/v1/books/:bookId/metadata/image/*",
    { preHandler: authHook },
    async (req, reply) => {
      const { bookId } = req.params;

      const book = await db
        .select({ coverPath: books.coverPath })
        .from(books)
        .where(eq(books.id, bookId))
        .get();

      if (!book?.coverPath) {
        return reply.code(404).send({ message: "No cover" });
      }

      const stream = storage.stream(book.coverPath);
      reply.header("Content-Type", "image/jpeg");
      return reply.send(stream);
    },
  );

  // Catch-all proxy — forward unhandled requests to Kobo store
  app.all<{ Params: { token: string; "*": string } }>(
    "/kobo/:token/*",
    { preHandler: authHook },
    async (req, reply) => {
      const wildcardPath = (req.params as any)["*"];
      const koboPath = `/${wildcardPath}`;

      const result = await proxyToKobo(
        req.method,
        koboPath,
        req.headers as Record<string, string>,
        req.body,
      );

      // Forward response headers
      for (const [key, value] of Object.entries(result.headers)) {
        if (key.toLowerCase() !== "transfer-encoding") {
          reply.header(key, value);
        }
      }

      return reply.code(result.status).send(result.body);
    },
  );
}
```

**Files:**

- [ ] **Step 1: Create the kobo tRPC router**

Create `packages/server/src/trpc/routers/kobo.ts`:

```typescript
import { eq } from "drizzle-orm";
import { router, protectedProcedure } from "../index.js";
import { koboSettings, koboSyncSnapshots, koboUpdateSettingsInput } from "@verso/shared";

export const koboRouter = router({
  getSettings: protectedProcedure.query(async ({ ctx }) => {
    const settings = await ctx.db
      .select({
        token: koboSettings.token,
        convertToKepub: koboSettings.convertToKepub,
        enabled: koboSettings.enabled,
      })
      .from(koboSettings)
      .where(eq(koboSettings.userId, ctx.user.sub))
      .get();

    return settings ?? null;
  }),

  enableSync: protectedProcedure.mutation(async ({ ctx }) => {
    // Check if already enabled
    const existing = await ctx.db
      .select({ id: koboSettings.id })
      .from(koboSettings)
      .where(eq(koboSettings.userId, ctx.user.sub))
      .get();

    if (existing) {
      // Re-enable
      await ctx.db
        .update(koboSettings)
        .set({ enabled: true, updatedAt: new Date().toISOString() })
        .where(eq(koboSettings.id, existing.id));
    } else {
      await ctx.db.insert(koboSettings).values({
        userId: ctx.user.sub,
        token: crypto.randomUUID(),
      });
    }

    const settings = await ctx.db
      .select({ token: koboSettings.token })
      .from(koboSettings)
      .where(eq(koboSettings.userId, ctx.user.sub))
      .get();

    return { token: settings!.token };
  }),

  disableSync: protectedProcedure.mutation(async ({ ctx }) => {
    // Delete snapshots
    await ctx.db
      .delete(koboSyncSnapshots)
      .where(eq(koboSyncSnapshots.userId, ctx.user.sub));
    // Delete settings
    await ctx.db
      .delete(koboSettings)
      .where(eq(koboSettings.userId, ctx.user.sub));
    return { success: true };
  }),

  regenerateToken: protectedProcedure.mutation(async ({ ctx }) => {
    const newToken = crypto.randomUUID();

    await ctx.db
      .update(koboSettings)
      .set({ token: newToken, updatedAt: new Date().toISOString() })
      .where(eq(koboSettings.userId, ctx.user.sub));

    // Clear snapshots — device will re-sync everything
    await ctx.db
      .delete(koboSyncSnapshots)
      .where(eq(koboSyncSnapshots.userId, ctx.user.sub));

    return { token: newToken };
  }),

  updateSettings: protectedProcedure
    .input(koboUpdateSettingsInput)
    .mutation(async ({ ctx, input }) => {
      const updates: Record<string, unknown> = { updatedAt: new Date().toISOString() };
      if (input.convertToKepub !== undefined) {
        updates.convertToKepub = input.convertToKepub;
      }

      await ctx.db
        .update(koboSettings)
        .set(updates)
        .where(eq(koboSettings.userId, ctx.user.sub));

      return { success: true };
    }),
});
```

```typescript
function KoboSection() {
  const { t } = useTranslation();
  const [showSetup, setShowSetup] = useState(false);
  const [copied, setCopied] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [confirmRegenerate, setConfirmRegenerate] = useState(false);
  const [confirmDisable, setConfirmDisable] = useState(false);

  const settingsQuery = trpc.kobo.getSettings.useQuery();

  const enableMut = trpc.kobo.enableSync.useMutation({
    onSuccess: () => { setError(""); settingsQuery.refetch(); },
    onError: (err) => setError(err.message),
  });

  const disableMut = trpc.kobo.disableSync.useMutation({
    onSuccess: () => { setError(""); setSuccess(""); settingsQuery.refetch(); },
    onError: (err) => setError(err.message),
  });

  const regenerateMut = trpc.kobo.regenerateToken.useMutation({
    onSuccess: () => {
      setSuccess(t("kobo.regenerated"));
      setError("");
      settingsQuery.refetch();
    },
    onError: (err) => setError(err.message),
  });

  const updateMut = trpc.kobo.updateSettings.useMutation({
    onSuccess: () => settingsQuery.refetch(),
    onError: (err) => setError(err.message),
  });

  const settings = settingsQuery.data;
  const syncUrl = settings ? `${window.location.origin}/kobo/${settings.token}` : "";

  const handleCopy = () => {
    navigator.clipboard.writeText(syncUrl);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="mt-10">
      <h2
        className="text-sm font-medium uppercase tracking-wider mb-2"
        style={{ color: "var(--text-dim)" }}
      >
        {t("kobo.title")}
      </h2>
      <p className="text-xs mb-4" style={{ color: "var(--text-faint)" }}>

                style={{ backgroundColor: "var(--card)", borderColor: "var(--border)", color: "var(--text-dim)" }}

                <p>{t("kobo.setup2")}</p>
                <p>{t("kobo.setup3")}</p>
                <p>{t("kobo.setup4")}<code className="font-mono text-[10px] px-1 py-0.5 rounded" style={{ backgroundColor: "var(--card)" }}>{syncUrl}</code></p>

      )}
```
